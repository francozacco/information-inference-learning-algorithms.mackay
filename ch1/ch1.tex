\section*{1 - Introduction to Information Theory}

\begin{proof}{\textbf{Exercise 1.2}}\\
When using $R_3$ we get an error if two or three bits are flipped in a set of
3 bits.
Then the probability of two bits being flipped is
\begin{align*}
    P(r=2~|~f, N=3) = \binom{3}{2}f^2(1 - f)^{3 - 2}
    = 3f^2(1 - f)
\end{align*}
And the probability of three bits being flipped is
\begin{align*}
    P(r=3~|~f, N=3) = \binom{3}{3}f^3(1 - f)^{3 - 3}
    = f^3
\end{align*}
Then the total probability of getting an error when using $R_3$ is
\begin{align*}
    P(2~|~f, 3) + P(3~|~f, 3) &= f^3 + 3f^2(1 - f)
    = 3f^2 - 2f^3
\end{align*}
We observe that $3f^2 - 2f^3$ is less than $f$ if $f < 0.5$, hence, the
error probability is reduced by the use of $R_3$.
\end{proof}
